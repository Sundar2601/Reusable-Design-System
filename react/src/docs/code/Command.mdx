# Command: React and Angular

Same component, both frameworks. Live previews are under **Components / Command** (React) and **Angular / Command**.

## React

```tsx
import { Command, CommandInput, CommandList, CommandEmpty, CommandGroup, CommandItem, CommandShortcut } from '@bounteous/ui'
```

```tsx
<Command>
  <CommandInput placeholder="Type a command or search..." />
  <CommandList>
    <CommandEmpty>No results found.</CommandEmpty>
    <CommandGroup heading="Suggestions">
      <CommandItem>Calendar</CommandItem>
      <CommandItem>Settings <CommandShortcut>⌘S</CommandShortcut></CommandItem>
    </CommandGroup>
  </CommandList>
</Command>
```

## Angular (ZardUI)

```ts
imports: [ZardCommandImports]  // from '@/shared/components/command'
```

```html
<z-command>
  <z-command-input placeholder="Type a command or search..." />
  <z-command-list>
    <z-command-option-group zLabel="Suggestions">
      <z-command-option zLabel="Calendar" zValue="calendar" />
      <z-command-option zLabel="Settings" zValue="settings" zShortcut="⌘S" />
    </z-command-option-group>
  </z-command-list>
</z-command>
```
